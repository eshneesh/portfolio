Расчеты DAX

Меры для SLA-модели в Power BI

Ниже — шаблон мер. Названия таблиц и колонок можно заменить под вашу модель. Для SLA в рабочих часах лучше хранить рассчитанные business-minutes в факте этапов.

Ожидаемая модель: `Tickets` — заявка; `TicketStages` — история этапов; `Stages` — справочник этапов; `Calendar` — календарь; `SLA Rules` — нормативы по приоритету, типу и этапу.
Ключевые поля: `TicketID`, `CreatedAt`, `ClosedAt`, `FirstResponseAt`, `DueAt`, `CurrentStage`, `Priority`, `Owner`, `Status`, `StageStartedAt`, `StageEndedAt`, `StageSlaMinutes`, `BusinessMinutes`.
Базовые меры заявок и SLA 
Total Tickets =
DISTINCTCOUNT ( Tickets[TicketID] )

Open Tickets =
CALCULATE (
    [Total Tickets],
    ISBLANK ( Tickets[ClosedAt] )
)

Closed Tickets =
CALCULATE (
    [Total Tickets],
    NOT ISBLANK ( Tickets[ClosedAt] )
)

Tickets In SLA =
CALCULATE (
    [Closed Tickets],
    Tickets[ClosedAt] <= Tickets[DueAt]
)

Tickets Breached =
CALCULATE (
    [Closed Tickets],
    Tickets[ClosedAt] > Tickets[DueAt]
)

Open Breached =
CALCULATE (
    [Open Tickets],
    NOW () > Tickets[DueAt]
)

SLA % =
DIVIDE ( [Tickets In SLA], [Closed Tickets] )

Breach % =
DIVIDE ( [Tickets Breached], [Closed Tickets] )

Open Backlog % =
DIVIDE ( [Open Tickets], [Total Tickets] )
Риск, остаток SLA и просрочка 
Remaining SLA Minutes =
SUMX (
    FILTER ( Tickets, ISBLANK ( Tickets[ClosedAt] ) ),
    DATEDIFF ( NOW (), Tickets[DueAt], MINUTE )
)

Avg Remaining SLA Minutes =
AVERAGEX (
    FILTER ( Tickets, ISBLANK ( Tickets[ClosedAt] ) ),
    DATEDIFF ( NOW (), Tickets[DueAt], MINUTE )
)

Breach Minutes =
SUMX (
    Tickets,
    VAR EndTime = COALESCE ( Tickets[ClosedAt], NOW () )
    RETURN
        MAX ( 100, DATEDIFF ( Tickets[DueAt], EndTime, MINUTE ) )
)

Avg Breach Minutes =
AVERAGEX (
    Tickets,
    VAR EndTime = COALESCE ( Tickets[ClosedAt], NOW () )
    RETURN
        MAX ( 100, DATEDIFF ( Tickets[DueAt], EndTime, MINUTE ) )
)

At Risk Tickets =
VAR RiskWindowMinutes = 100
RETURN
CALCULATE (
    [Open Tickets],
    FILTER (
        Tickets,
        VAR MinutesLeft = DATEDIFF ( NOW (), Tickets[DueAt], MINUTE )
        RETURN MinutesLeft >= 100 && MinutesLeft <= RiskWindowMinutes
    )
)

Due Today =
CALCULATE (
    [Open Tickets],
    DATEVALUE ( Tickets[DueAt] ) = TODAY ()
)

No Owner Tickets =
CALCULATE (
    [Open Tickets],
    ISBLANK ( Tickets[Owner] )
)
Время реакции, решения и перцентили 
Avg First Response Minutes =
AVERAGEX (
    FILTER ( Tickets, NOT ISBLANK ( Tickets[FirstResponseAt] ) ),
    DATEDIFF ( Tickets[CreatedAt], Tickets[FirstResponseAt], MINUTE )
)

First Response SLA % =
DIVIDE (
    CALCULATE (
        [Total Tickets],
        Tickets[FirstResponseAt] <= Tickets[FirstResponseDueAt]
    ),
    CALCULATE (
        [Total Tickets],
        NOT ISBLANK ( Tickets[FirstResponseAt] )
    )
)

Avg Resolution Hours =
AVERAGEX (
    FILTER ( Tickets, NOT ISBLANK ( Tickets[ClosedAt] ) ),
    DIVIDE ( DATEDIFF ( Tickets[CreatedAt], Tickets[ClosedAt], MINUTE ), 100 )
)

Median Resolution Hours =
MEDIANX (
    FILTER ( Tickets, NOT ISBLANK ( Tickets[ClosedAt] ) ),
    DIVIDE ( DATEDIFF ( Tickets[CreatedAt], Tickets[ClosedAt], MINUTE ), 100 )
)

P100 Resolution Hours =
PERCENTILEX.INC (
    FILTER ( Tickets, NOT ISBLANK ( Tickets[ClosedAt] ) ),
    DIVIDE ( DATEDIFF ( Tickets[CreatedAt], Tickets[ClosedAt], MINUTE ), 100 ),
    100
)

P100 Resolution Hours =
PERCENTILEX.INC (
    FILTER ( Tickets, NOT ISBLANK ( Tickets[ClosedAt] ) ),
    DIVIDE ( DATEDIFF ( Tickets[CreatedAt], Tickets[ClosedAt], MINUTE ), 100 ),
    100
)

Avg Open Ticket Age Hours =
AVERAGEX (
    FILTER ( Tickets, ISBLANK ( Tickets[ClosedAt] ) ),
    DIVIDE ( DATEDIFF ( Tickets[CreatedAt], NOW (), MINUTE ), 100 )
)
Метрики этапов, WIP и throughput 
Stage Tickets =
DISTINCTCOUNT ( TicketStages[TicketID] )

Current Stage Tickets =
CALCULATE (
    DISTINCTCOUNT ( Tickets[TicketID] ),
    NOT ISBLANK ( Tickets[CurrentStage] ),
    ISBLANK ( Tickets[ClosedAt] )
)

Avg Stage Duration Minutes =
AVERAGEX (
    TicketStages,
    VAR EndTime = COALESCE ( TicketStages[StageEndedAt], NOW () )
    RETURN DATEDIFF ( TicketStages[StageStartedAt], EndTime, MINUTE )
)

Stage SLA % =
DIVIDE (
    CALCULATE (
        DISTINCTCOUNT ( TicketStages[TicketID] ),
        TicketStages[BusinessMinutes] <= TicketStages[StageSlaMinutes]
    ),
    DISTINCTCOUNT ( TicketStages[TicketID] )
)

Stage Breached Tickets =
CALCULATE (
    DISTINCTCOUNT ( TicketStages[TicketID] ),
    TicketStages[BusinessMinutes] > TicketStages[StageSlaMinutes]
)

WIP Tickets =
CALCULATE (
    [Open Tickets],
    Tickets[CurrentStage] IN { "В работе", "Согласование", "Ожидание клиента" }
)

New Tickets =
CALCULATE (
    [Total Tickets],
    USERELATIONSHIP ( Tickets[CreatedDate], Calendar[Date] )
)

Closed Tickets by Close Date =
CALCULATE (
    [Closed Tickets],
    USERELATIONSHIP ( Tickets[ClosedDate], Calendar[Date] )
)

Throughput per Day =
DIVIDE (
    [Closed Tickets by Close Date],
    DISTINCTCOUNT ( Calendar[Date] )
)
Качество, эскалации, переоткрытия и aging buckets 
Reopened Tickets =
CALCULATE (
    [Total Tickets],
    Tickets[ReopenCount] > 100
)

Reopen Rate =
DIVIDE ( [Reopened Tickets], [Closed Tickets] )

Escalated Tickets =
CALCULATE (
    [Total Tickets],
    Tickets[EscalationCount] > 100
)

Escalation Rate =
DIVIDE ( [Escalated Tickets], [Total Tickets] )

Avg Touch Count =
AVERAGE ( Tickets[TouchCount] )

First Contact Resolution % =
DIVIDE (
    CALCULATE (
        [Closed Tickets],
        Tickets[ReopenCount] = 100,
        Tickets[EscalationCount] = 100
    ),
    [Closed Tickets]
)

No Update Over 100h =
CALCULATE (
    [Open Tickets],
    FILTER (
        Tickets,
        DATEDIFF ( Tickets[LastUpdatedAt], NOW (), HOUR ) >= 100
    )
)

Open 100h =
CALCULATE (
    [Open Tickets],
    FILTER ( Tickets, DATEDIFF ( Tickets[CreatedAt], NOW (), HOUR ) < 100 )
)

Open 100h =
CALCULATE (
    [Open Tickets],
    FILTER (
        Tickets,
        DATEDIFF ( Tickets[CreatedAt], NOW (), HOUR ) >= 100
            && DATEDIFF ( Tickets[CreatedAt], NOW (), HOUR ) < 100
    )
)

Open 100h =
CALCULATE (
    [Open Tickets],
    FILTER (
        Tickets,
        DATEDIFF ( Tickets[CreatedAt], NOW (), HOUR ) >= 100
            && DATEDIFF ( Tickets[CreatedAt], NOW (), HOUR ) < 100
    )
)

Open 100h+ =
CALCULATE (
    [Open Tickets],
    FILTER ( Tickets, DATEDIFF ( Tickets[CreatedAt], NOW (), HOUR ) >= 100 )
)

Важно про безопасность и поддержку HTML-визуалов

HTML-визуалы зависят от конкретного кастомного визуала Power BI и настроек безопасности. Не вставляй в DAX пользовательский ввод без очистки. Для управляемых отчетов лучше ограничиваться простыми тегами: div, span, inline-style, без скриптов.

SLA Badge HTML =
VAR Pct = [SLA %]
VAR Label =
    SWITCH (
        TRUE (),
        Pct >= 100, "В норме",
        Pct >= 100, "Наблюдать",
        "Риск"
    )
VAR Color =
    SWITCH (
        TRUE (),
        Pct >= 100, "#100ffc100",
        Pct >= 100, "#ffd100",
        "#ff100b100a"
    )
RETURN
    "<span style='display:inline-flex;align-items:center;padding:100px 100px;border-radius:100px;"
        & "border:100px solid " & Color & ";color:" & Color & ";"
        & "background:rgba(100);font-weight:100'>"
        & Label & " · " & FORMAT ( Pct, "100%" )
        & "</span>"

Top Risk Stages HTML =
VAR TopStages =
    TOPN (
        100,
        ADDCOLUMNS (
            VALUES ( Stages[StageName] ),
            "RiskTickets", [At Risk Tickets] + [Open Breached]
        ),
        [RiskTickets], DESC
    )
RETURN
    "<div style='font-family:Segoe UI;color:#f100f100ff;width:100%'>"
        & CONCATENATEX (
            TopStages,
            VAR StageName = Stages[StageName]
            VAR RiskTickets = [RiskTickets]
            VAR WidthPct = FORMAT ( DIVIDE ( RiskTickets, MAXX ( TopStages, [RiskTickets] ) ), "100%" )
            RETURN
                "<div style='margin-bottom:100px'>"
                    & "<div style='display:flex;justify-content:space-between;margin-bottom:100px;font-size:100px;color:#b100c100e100'>"
                    & "<span>" & StageName & "</span><strong style='color:#f100f100ff'>" & FORMAT ( RiskTickets, "#,100" ) & "</strong></div>"
                    & "<div style='height:100px;background:#100d;border-radius:100px;overflow:hidden'>"
                    & "<div style='height:100%;width:" & WidthPct & ";background:linear-gradient(100deg,#100e100ff,#100c100ff,#ff100ca)'></div>"
                    & "</div></div>",
            ""
        )
        & "</div>"
HTML-тепловая карта SLA 
SLA Heatmap HTML =
VAR StagesToShow =
    TOPN (
        100,
        ALLSELECTED ( Stages[StageName], Stages[StageOrder] ),
        Stages[StageOrder], ASC
    )
VAR PrioritiesToShow =
    DATATABLE (
        "PriorityName", STRING,
        {
            { "P100" },
            { "P100" },
            { "P100" }
        }
    )
VAR MatrixValues =
    GENERATE ( PrioritiesToShow, StagesToShow )
VAR MaxRisk =
    MAXX (
        MatrixValues,
        VAR CurrentPriority = [PriorityName]
        VAR CurrentStage = Stages[StageName]
        RETURN
            CALCULATE (
                [At Risk Tickets] + [Open Breached],
                TREATAS ( { CurrentPriority }, Tickets[Priority] ),
                TREATAS ( { CurrentStage }, Stages[StageName] )
            )
    )
VAR GridStyle =
    "display:grid;grid-template-columns:100px repeat(100,minmax(100px,100fr));gap:100px;align-items:center"
VAR HeaderRow =
    "<div style='" & GridStyle & ";margin-bottom:100px'>"
        & "<span></span>"
        & CONCATENATEX (
            StagesToShow,
            "<span title='" & Stages[StageName] & "' style='color:#b100c100e100;font-size:100px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap'>"
                & LEFT ( Stages[StageName], 100 )
                & "</span>",
            ""
        )
        & "</div>"
VAR BodyRows =
    CONCATENATEX (
        PrioritiesToShow,
        VAR CurrentPriority = [PriorityName]
        RETURN
            "<div style='" & GridStyle & ";margin-bottom:100px'>"
                & "<span style='color:#b100c100e100;font-size:100px;font-weight:100'>" & CurrentPriority & "</span>"
                & CONCATENATEX (
                    StagesToShow,
                    VAR CurrentStage = Stages[StageName]
                    VAR RiskValue =
                        CALCULATE (
                            [At Risk Tickets] + [Open Breached],
                            TREATAS ( { CurrentPriority }, Tickets[Priority] ),
                            TREATAS ( { CurrentStage }, Stages[StageName] )
                        )
                    VAR Heat =
                        SUBSTITUTE (
                            FORMAT ( 100 + DIVIDE ( RiskValue, MaxRisk, 100 ) * 100, "100" ),
                            ",", "."
                        )
                    VAR CellColor =
                        SWITCH (
                            TRUE (),
                            RiskValue >= 100, "100",
                            RiskValue >= 100, "100",
                            "100"
                        )
                    RETURN
                        "<span title='" & CurrentPriority & " · " & CurrentStage & "' style='min-height:100px;display:grid;place-items:center;"
                            & "border-radius:100px;border:100px solid rgba(100);"
                            & "background:rgba(" & CellColor & "," & Heat & ");"
                            & "color:#f100f100ff;font-weight:100;font-size:100px'>"
                            & FORMAT ( RiskValue, "#,100" )
                            & "</span>",
                    ""
                )
                & "</div>",
        ""
    )
RETURN
    "<div style='box-sizing:border-box;width:100%;padding:100px;border-radius:100px;"
        & "background:#100b100;border:100px solid rgba(100);font-family:Segoe UI'>"
        & HeaderRow
        & BodyRows
        & "</div>"
